    function formatPrice(price) {
        if (price >= 10000000) return (price / 1000000).toFixed(1) + " млн ₽";
        return price.toLocaleString('ru-RU') + " ₽";
    }

    function renderHouses(filterLocation = "Москва", filterPrice = "any", filterRooms = "any") {
        let filtered = housesData.filter(h => h.location.toLowerCase().includes(filterLocation.toLowerCase()));
        if (filterPrice !== "any") filtered = filtered.filter(h => h.price <= parseInt(filterPrice) * 1000000);
        if (filterRooms !== "any") filtered = filtered.filter(h => h.rooms >= parseInt(filterRooms));
        const container = document.getElementById('housesContainer');
        const countSpan = document.getElementById('houseCount');
        if (!container) return;
        
        if (countSpan) {
            countSpan.textContent = filtered.length + " из " + housesData.length + " объектов";
        }
        
        if (filtered.length === 0) { 
            container.innerHTML = '<div style="grid-column:1/-1; text-align:center; padding:60px;">🏡 Домов не найдено. Попробуйте изменить параметры поиска.</div>'; 
            return; 
        }
        
        container.innerHTML = filtered.map(house => `
            <div class="house-card" onclick="showHouseDetails(${house.id})">
                <div class="card-svg">
                    ${house.hot ? '<div class="badge-hot">🔥 Горячее предложение</div>' : ''}
                    ${house.price > 50000000 ? '<div class="badge-new">⭐ Премиум</div>' : ''}
                    <div style="font-size:70px;">${house.style}</div>
                </div>
                <div class="card-content">
                    <div class="price">${formatPrice(house.price)}</div>
                    <div class="address"><i class="fas fa-map-marker-alt"></i> ${house.location}</div>
                    <div class="details"><span><i class="fas fa-bed"></i> ${house.rooms} комн.</span><span><i class="fas fa-ruler-combined"></i> ${house.area} м²</span></div>
                    <div class="card-footer"><span class="agent"><i class="fas fa-user"></i> Агент: ${house.agent}</span><span style="color:#1f8a4c;">Подробнее →</span></div>
                </div>
            </div>
        `).join('');
    }

        }
    }

    const chatWindow = document.getElementById('chatWindow');

    const closeChatBtn = document.getElementById('closeChatBtn');

    const chatInput = document.getElementById('chatInput');

    function openChat() { chatWindow.style.display = 'flex'; }
    function closeChat() { chatWindow.style.display = 'none'; }

    closeChatBtn.addEventListener('click', closeChat);
    
    function addMessage(text, isUser) {
        const div = document.createElement('div');
        div.className = `message ${isUser ? 'user-message' : 'bot-message'}`;
        div.textContent = text;
        chatMessages.appendChild(div);
        chatMessages.scrollTop = chatMessages.scrollHeight;
    }
    
    function showTyping() {
        const typingDiv = document.createElement('div');
        typingDiv.className = 'typing-indicator';
        typingDiv.id = 'typingIndicator';
        typingDiv.textContent = '✍️ AI печатает...';
        chatMessages.appendChild(typingDiv);
        chatMessages.scrollTop = chatMessages.scrollHeight;
    }
    
    function removeTyping() {
        const typing = document.getElementById('typingIndicator');
        if (typing) typing.remove();
    }
    
    async function sendMessage() {
        const message = chatInput.value.trim();
        if (!message) return;
        addMessage(message, true);
        chatInput.value = '';
        
        showTyping();
        
        setTimeout(() => {
            removeTyping();
            const response = getAIResponse(message);
            addMessage(response, false);
        }, 500 + Math.random() * 400);
    }
